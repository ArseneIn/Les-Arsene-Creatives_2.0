<script>
    function bulkEntryData() {
        return {
            sessions: {
                {% for student in students %}
                {{ student.id }}: {
                    wpm: '{{ student.current_session.wpm|default:"" }}',
                    accuracy: '{{ student.current_session.accuracy|default:"" }}',
                    changed: false
                },
                {% endfor %}
            },
            changedCount: 0,

            markChanged(studentId) {
                if (!this.sessions[studentId].changed) {
                    this.sessions[studentId].changed = true;
                    this.updateChangedCount();
                }
            },

            updateChangedCount() {
                this.changedCount = Object.values(this.sessions).filter(s =>
                    s.changed && (s.wpm || s.accuracy)
                ).length;
            },

            submitAllScores() {
                const scoresToSubmit = [];

                for (const [studentId, session] of Object.entries(this.sessions)) {
                    if (session.changed && (session.wpm || session.accuracy)) {
                        scoresToSubmit.push({
                            student_id: studentId,
                            wpm: parseFloat(session.wpm) || 0,
                            accuracy: parseFloat(session.accuracy) || 0
                        });
                    }
                }

                if (scoresToSubmit.length === 0) {
                    alert('No scores to submit');
                    return;
                }

                // Submit via form
                const form = document.createElement('form');
                form.method = 'POST';
                form.action = '{% url "bulk_entry" %}';

                const csrfToken = document.querySelector('[name=csrfmiddlewaretoken]').value;
                form.innerHTML = `
                        <input type="hidden" name="csrfmiddlewaretoken" value="${csrfToken}">
                        <input type="hidden" name="scores_data" value='${JSON.stringify(scoresToSubmit)}'>
                        <input type="hidden" name="week" value="{{ week_number }}">
                        <input type="hidden" name="program" value="{{ selected_program_id }}">
                    `;

                document.body.appendChild(form);
                form.submit();
            }
        }
    }
</script>
